\beginsong{Isus — izvor de mângâiere}
\beginverse
Isus — izvor de mângâiere,
La Tine vin încrezător;
Sunt copleșit de-o grea durere
Și mistuit de-al ei fior.
\endverse

\beginchorus
Isus, Isus, azi vin la Tine
Cu-adânci suspine,
La Tine vin;
Iubirea Ta să mă lumine
Și să-mi aduc-al Tău alin!
\endchorus

\beginverse
Isus — izvor de bucurie,
Sunt întristat azi de vrăjmași;
Să mă păzești, să-mi dai tărie
Și-n mâna lor să nu mă lași!
\endverse

\beginverse
Isus — izvor de har și pace,
Adâncă pace Tu să-mi dai;
Chiar carnea de mi s-ar desface,
În duhul meu e cer, e rai!
\endverse

\beginverse
Ascunde viața mea în Tine,
Atunci când lupte grele vin;
Tu, numai Tu să fii în mine,
Izvorul harurilor plin!
\endverse
\endsong
