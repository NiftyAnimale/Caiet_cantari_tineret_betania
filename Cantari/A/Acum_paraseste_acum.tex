\beginsong{Acum, părăsește acum}

\begin{verse}
	Acum, părăsește acum
	gândul tău, gând murdar,
	ce-i în inima ta.
	Senin sufletul îți va fi,
	dacă vei părăsi răul din viața ta.
\end{verse}

\begin{chorus}
	/: Vino la Isus, viață-ți aduce,
	Din moarte El a înviat!
	Pace, iertare și-n inima ta,
	Suferința nu va exista! :/
\end{chorus}

\begin{verse}
	Să vii la Isus chiar acum,
	să pășești pe al Său drum
	și să-I dai inima.
	Isus te primește cum ești,
	dar să nu zăbovești,
	Mântuire-ți va da!
\end{verse}

\endsong
