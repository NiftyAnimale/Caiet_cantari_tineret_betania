\beginsong{Domn minunat, ești lângă mine}

\beginverse
Domn minunat ești lângă mine,
În orice zi eu sunt cu Tine
Căci mâna Ta nu m-a lăsat
Să rătăcesc.
\endverse

\beginchorus
Ia-mă, schimbă-mă, folosește-mă,
Viața mea e în mâna Ta
Glasul Tău vreau mereu,
Să-L am în sufletul meu
Inima mea, Doamne, e a Ta!
\endchorus

\beginverse
Cu gingășie mă strigi pe nume,
Duhul Tău Sfânt îmi dă mângâiere
Și-aduce pace în sufletul meu,
Căci mă iubești.
\endverse

\beginverse
Sunt copleșit de-a Ta iubire,
M-ai cucerit pentru vecie,
O, Domnul meu, vreau
Să Îți mulțumesc.
\endverse

\beginverse
Domn minunat ești lângă mine,
În orice zi eu cât sunt cu Tine,
Cât timp trăiesc prin viața mea,
Vreau să Te slăvesc.
\endverse

\endsong
