\beginsong{De dragul meu Isus a venit,}

\beginverse
De dragul meu Isus a venit,
În locul meu El a murit,
/: Să ia păcatul lumii,
Să-l ducă la Calvar;
Ce mare har, ce mare har! :/
\endverse

\beginchorus
Golgota, Golgota,
Loc de chin și suferință.
Golgota, Golgota,
Acolo a fost biruință.
\endchorus

\beginverse
De dragul tău Isus S-a jertfit,
Pentru tine ceru-a părăsit,
Să-ți dea în dar iertare
Și viață veșnică;
/: Ce mare har, ce mare har! :/
\endverse

\beginchorus
Povara omenirii Hristos a purtat,
Din palme I-a curs sângele nevinovat.
Hristos a murit, a fost crucificat,
Dar a treia zi a-nviat.
\endchorus

\endsong
