\beginsong{Doamne, Îți dau primul loc în viața mea}

\beginverse
Doamne, Îți dau
Primul loc în viața mea,
Să fii primul gând
Când ochii îi deschid,
Și în inima mea
Cerul să fie voia Ta,
Și legile Tale să-mi fie stelele.
Seara gândind la-l Tău Cuvânt,
Ochii să-nchid.
\endverse

\beginverse
Doamne, Tu ai primul
Loc în viața mea,
Tu ești primul gând
Când ochii îi deschid,
Și în inima mea
Cerul este voința Ta,
Și cuvintele Tale sunt ca stelele,
Seara gândind la-l Tău Cuvânt,
Ochii închid.
\endverse

\endsong
