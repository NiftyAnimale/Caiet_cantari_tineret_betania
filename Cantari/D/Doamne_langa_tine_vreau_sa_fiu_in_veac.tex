\beginsong{Doamne, lângă Tine vreau să fiu în veac}

\beginverse
Doamne, lângă Tine vreau să fiu în veac;
Doamne, fără Tine nici un pas să fac.
\endverse

\beginchorus
De la Tine, Domnul meu,
Vreau să-nvăț mereu.
Cum faci Tu să fac și eu,
Cum stai Tu și eu să stau,
Ce vrei Tu și eu să vreau,
Crucea Ta s-o iau.
\endchorus

\beginverse
Doamne, pentru Tine viața vreau să-mi pun,
Crucea Ta-i frumoasă, jugul Tău e bun.
\endverse

\beginverse
Doamne, pe-a Ta cale vreau să merg mereu
Până voi ajunge lângă Dumnezeu.
\endverse

\endsong
