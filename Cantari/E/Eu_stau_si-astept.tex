\beginsong{Eu stau și-aștept să se-ntâmple ceva}
\beginverse
Eu stau și-aștept să se-ntâmple ceva,
Ceva să-mi schimbe azi viața mea.
Te văd cum cobori muntele
Și toți vor să se-atingă de haina Ta.
Tu ești Cel ce face minuni
Și știu, răspunsul meu e-n mâna Ta.
Da, vin, mă apropii și mă închin
Și-n teamă spun cu vocea mea:
\endverse

\beginchorus
Doamne, dacă vrei, Tu poți!
Doamne, dacă vrei, Tu poți!
Da, Doamne, dacă vrei, Tu poți
Să m-atingi cu Mâna Ta,
Să atingi azi viața mea.
\endchorus

\beginverse
Azi viața mea, Isuse, atinge-o Tu,
Atinge viața mea, aici, acum, aici, acum;
Cu mâna Ta, Te rog, Isuse,
Isuse, atinge viața mea.
\endverse
\endsong
