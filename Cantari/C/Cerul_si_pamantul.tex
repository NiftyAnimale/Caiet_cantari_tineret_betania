\beginsong{Cerul și pământul (Hristos se naște)}
\beginverse
/: Cerul și pământul :/
În cântec răsună,
/: Îngeri și oameni :/
Cântă împreună.
\endverse

\beginchorus
/: Hristos se naște,
Duhul Sfânt coboară,
Îngerii cântă, magii Îl adoră,
Păstori aleargă, ieslea o-mpresoară,
Mari minuni se întâmplară. :/
\endchorus

\beginverse
/: Din răsărit vin :/
Magi cu bucurie,
/: Cu dar de smirnă, :/
Aur și tămâie.
\endverse

\beginverse
/: În Betleem :/
Ce mare minune:
/: Vergură curată :/
Fiu născut în lume.
\endverse

\beginverse
/: Hristos se naște, :/
Veniți la închinare,
/: Cu vesel suflet, :/
Veselă cântare!
\endverse
\endsong
