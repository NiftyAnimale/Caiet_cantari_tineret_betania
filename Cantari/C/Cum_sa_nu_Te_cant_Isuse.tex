
\beginsong{Cum să nu Te cânt, Isuse}[by={Traian Dorz}]

\beginverse
Cum să nu Te cânt, Isuse,
cum să nu Te cânt?
Cum să nu-Ți slăvesc cu lacrimi
Numele Tău sfânt.
\endverse

\beginchorus
C-am strigat spre Tine, Doamne,
și m-ai ascultat,
bucuria și lumina
mie mi le-ai dat.
\endchorus

\beginverse
Cum să tac, o, cum,
că-mi vine, lumii-ntregi să-i spun
cât de bun ai fost cu mine,
cât, o, cât de bun!
\endverse

\beginverse
Unde-i lacrima și glasul,
inima de-ajuns,
să-Ți slăvesc cu ele mila
care mi-a răspuns?
\endverse

\beginverse
Până-n veci voi spune, Doamne,
cât de-adevărat,
Ți-e Cuvântul și Iubirea
ce m-au ascultat.
\endverse

\endsong
