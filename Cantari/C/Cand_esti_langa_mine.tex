\beginsong{Când ești lângă mine}

\beginverse
Când ești lângă mine,
Nu mă tem de niciun rău;
Căci mă-ncred în Tine,
Mare Dumnezeu!
\endverse

\beginchorus
Numai lângă Tine,
Doamne, mă simt bine,
Știu că Tu mă ocrotești.
Și-n grea încercare,
Când furtuna-i mare,
Lângă mine Tu pășești!
Numai lângă Tine,
Doamne, mă simt bine,
Știu că Tu mă ocrotești.
Și-n grea încercare,
Când furtuna-i mare,
Lângă mine-alături ești!
\endchorus

\beginverse
Valurile vieții,
Barca când mi-au clătinat,
Ca și altădată,
Tu, Isuse, le-ai mustrat!
\endverse

\beginverse
Meriți pe vecie,
Doamne, să fii lăudat!
Căci, din grea robie,
Doar Tu m-ai eliberat!
\endverse

\beginverse
O nădejde vie
În inimă mi-ai sădit;
Cred, deci, cu tărie,
Doamne, că m-ai mântuit!
\endverse

\endsong
