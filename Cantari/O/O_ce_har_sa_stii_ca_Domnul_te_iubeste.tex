\beginsong{O, ce har să știi că Domnul te iubește}
\beginverse
O, ce har să știi că Domnul te iubește,
O, ce har să știi că ești în brațul Său!
Chiar când simți că tot în jur se prăbușește,
Pace tu să ai în inimă mereu!
\endverse

\beginchorus
Să ai pace în mijlocul necazului greu,
Să ai pace, că tu ești fiu de Dumnezeu!
Când cei din jur te vor privi,
Cu pacea ta îi vei uimi!
Fii plin de pace, vei birui!
\endchorus

\beginverse
O, ce har să știi că Domnul te iubește,
O, ce har să știi că Lui îi aparții,
Și să simți în orice zi că-ți dăruiește
Pace în necaz și multe bucurii!
\endverse

\beginverse
O, ce har să știi că Domnul te iubește,
O, ce har să știi că ești copilul Său;
Zi de zi prezent fiind, te însoțește
Și-ți dă pace în necazul cel mai greu!
\endverse
\endsong
