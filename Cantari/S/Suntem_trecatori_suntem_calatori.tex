\beginsong{Suntem trecători, suntem călători}
\beginverse
Suntem trecători, suntem călători,
Pornim înspre țara de vis;
Oriunde am fi, oriunde ne-am găsi,
Nu va fi nicicând ca-n Paradis.
\endverse

\beginchorus
/: Acolo ne este țara, promisă de Isus,
Cerul sfânt, patria dorită,
Țara cea de sus. :/
\endchorus

\beginverse
Suntem trecători, suntem călători,
Mai avem puțin de mers
Și în curând, de pe pământ,
Vom pleca spre Univers.
\endverse

\beginverse
Suntem trecători, suntem călători,
Și cântecul acesta s-a sfârșit;
Curând pe nori, acolo în splendori,
Dumnezeu ne va răpi.
\endverse
\endsong
