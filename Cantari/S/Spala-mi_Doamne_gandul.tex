\beginsong{Spală-mi, Doamne, gândul}[by={Costache Ioanid}]
\beginverse
Spală-mi, Doamne, gândul,
Spală-mi, Doamne, viața,
Vreau să fiu curat și plăcut în fața Ta;
Numai Tu urci zorii
Și împrăștii ceața,
Numai Tu ești scutul și răsplata mea.
\endverse

\beginchorus
/: Spală-mi, Doamne, gândul,
Spală-mi, Doamne, viața
Și mă fă asemeni cu Tine mai mult;
Numai Tu urci zorii
Și împrăștii ceața,
Numai Tu ești scutul sufletului meu. :/
\endchorus

\beginverse
Spală-mi, Doamne, gândul,
Spală-mi, Doamne, viața,
Ia din mintea mea orice sfat al celui rău;
Fă să fiu ca Tine,
Să-Ți ascult povața,
Vreau să pot privi mereu în ochii Tăi.
\endverse

\beginverse
Spală-mi, Doamne, gândul,
Spală-mi, Doamne, viața,
Când troiene reci fac să-mi pară noaptea grea;
Adu primăvara și topește gheața
Ca să merg spre cer
Doar pe ea, pe calea Ta.
\endverse
\endsong
