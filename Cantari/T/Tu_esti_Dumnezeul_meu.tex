\beginsong{Tu ești Dumnezeul meu}
\beginverse
Tu ești Dumnezeul meu,
Tu-mi ești veșnica plăcere!
Crește-mă, prin har, mereu,
În lumină și-n putere,
Ca atunci când trec prin greu,
Să-Ți plinesc divina vrere!
/: Te laud, Dumnezeul meu! :/
\endverse

\beginverse
Tu ești Dumnezeul meu,
Creatorule a toate!
Te voi înălța mereu
Cu cântări înflăcărate!
Dă-mi Tu har când sunt la greu,
Să n-am jertfe întinate!
/: Te laud, Dumnezeul meu! :/
\endverse

\beginverse
Tu ești Dumnezeul meu,
Dragoste nemăsurată!
Laud Numele-Ți mereu,
Pacea Ta îmbelșugată!
Și când mi-e urcușul greu,
Sunt în mâna Ta de Tată!
/: Te laud, Dumnezeul meu! :/
\endverse
\endsong
