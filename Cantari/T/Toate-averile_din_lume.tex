\beginsong{Toate-averile din lume}
\beginverse
Toate-averile din lume
Sunt un fum, sunt un fum,
Tot ce ai poți pierde astăzi,
Chiar acum, chiar acum.
Singura avere care
N-are-apus, n-are-apus,
E Isus, e Isus, e Isus, Isus!
\endverse

\beginchorus
Să-L câștigi, să-L câștigi pe El,
Vinde casă, uită masă,
Totul lasă,
Numai să-L câștigi,
Numai să-L câștigi pe El!
\endchorus

\beginverse
Nu-i mărgăritar pe lume,
Aur nu-i, aur nu-i
Și nimic nu-i să mai aibă
Prețul Lui, prețul Lui.
Lucrul care trebuiește
Mai presus, mai presus,
E Isus, e Isus, e Isus, Isus!
\endverse

\beginverse
Toată slava lumii piere,
Cum e scris, cum e scris,
Toate-s trudă și durere,
Scrum și vis, scrum și vis.
Doar o slavă ține veșnic
Jos și sus, jos și sus,
E Isus, e Isus, e Isus, Isus!
\endverse
\endsong
