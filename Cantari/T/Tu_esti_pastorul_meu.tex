\beginsong{Tu ești păstorul meu, al sufletului meu}
\beginverse
Tu ești păstorul meu, al sufletului meu,
Așez viața mea azi în mâna Ta,
Astăzi hotărăsc cu Tine să trăiesc,
Și-n fiecare zi al Tău voi fi.
\endverse

\beginchorus
/: Nu mai vreau să rătăcesc,
Azi la cruce mă opresc,
Ție, pe deplin, Isus, mă dăruiesc,
Curățește-mi inima,
Umple-o de prezența Ta,
Și împărățește Tu pe veci în ea. :/
\endchorus

\beginverse
Tu ești păstorul meu, al sufletului meu,
Așez viața mea azi în mâna Ta,
Astăzi hotărăsc cu Tine să trăiesc,
Și vreau să Îți spun: Te iubesc!
\endverse

\beginchorus
/: Chiar de-ar fi prin valea morții,
Sau prin încercări să trec,
Știu că El, Păstorul meu, mă va păzi,
Pot să-nfrunt orice munte,
Orice val și orice greu,
Căci păstorul meu e însuși Dumnezeu. :/
\endchorus
\endsong
